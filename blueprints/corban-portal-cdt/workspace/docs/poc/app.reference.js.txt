  const fmtDate = v => { if (!v) return ''; const [y, m, d] = v.split('-'); return d + '/' + m + '/' + y; };
  function addDays(isoStr, n) {
    if (!isoStr || !n) return '';
    const [y, m, d] = isoStr.split('-').map(Number);
    const dt = new Date(Date.UTC(y, m - 1, d + n));
    return dt.getUTCFullYear() + '-' + String(dt.getUTCMonth() + 1).padStart(2, '0') + '-' + String(dt.getUTCDate()).padStart(2, '0');
  }

  const DEC = ['', '', '', 'TREINTA', 'CUARENTA', 'CINCUENTA', 'SESENTA', 'SETENTA', 'OCHENTA', 'NOVENTA'];
  const CEN = ['', 'CIENTO', 'DOSCIENTOS', 'TRESCIENTOS', 'CUATROCIENTOS', 'QUINIENTOS', 'SEISCIENTOS', 'SETECIENTOS', 'OCHOCIENTOS', 'NOVECIENTOS'];
  function menorMil(n) {
    if (n === 100) return 'CIEN';
    const c = Math.floor(n / 100), r = n % 100;
    let s = c ? CEN[c] : '';
    if (r) {
      if (s) s += ' ';
      if (r < 30) s += U[r];
      else { const d = Math.floor(r / 10), u = r % 10; s += DEC[d] + (u ? ' Y ' + U[u] : ''); }
    }
    return s;
  }
  function bajoMillon(n) {
    const miles = Math.floor(n / 1000), r = n % 1000, p = [];
    if (miles) p.push(miles === 1 ? 'MIL' : menorMil(miles) + ' MIL');
    if (r) p.push(menorMil(r));
    return p.join(' ');
  }
  function enLetras(n) {
    if (!n || n >= 1e12) return '';
    const mm = Math.floor(n / 1e6), rest = n % 1e6;
    let s = '';
    if (mm) s = mm === 1 ? 'UN MILLÓN' : (mm < 1000 ? menorMil(mm) : bajoMillon(mm)) + ' MILLONES';
    if (rest) s += (s ? ' ' : '') + bajoMillon(rest);
    if (mm && !rest) s += ' DE';
    return s;
  }
  const valorLetras = n => n ? enLetras(n) + ' PESOS M/CTE' : '';

  /* ---------- Derivados ---------- */
  function derive() {
    const D = Object.assign({}, data);
    const dias = num(data.plazo_dias);
    D.fecha_venc = addDays(data.fecha_apertura, dias);
    D.plazo_meses = dias ? String(Math.round(dias / 30)) : '';
    D.plazo_txt = dias ? dias + ' días' : '';
    D.valor = num(data.valor) || '';
    D.valor_letras = valorLetras(num(data.valor));
    D.nombre_completo = [data.nombre1, data.nombre2, data.apellido1, data.apellido2].filter(Boolean).join(' ');
    D.lugar_nac = [data.nac_ciudad, data.nac_pais].filter(Boolean).join(', ');
    D.ciudad_depto_res = [data.ciudad_res, data.depto_res].filter(Boolean).join(', ');
    D.ciudad_depto_trab = [data.ciudad_trab, data.depto_trab].filter(Boolean).join(', ');
    D.contrato_txt = label('contrato', data.contrato);
    D.periodicidad_txt = data.periodicidad === 'vencimiento' ? 'Al vencimiento' : label('periodicidad', data.periodicidad);
    const ing = num(data.ingresos), oi = num(data.otros_ingresos);
    D.total_ingresos = (ing + oi) || '';
    D.patrimonio = (num(data.activos) || num(data.pasivos)) ? (num(data.activos) - num(data.pasivos)) : '';
    ['ingresos', 'otros_ingresos', 'egresos', 'activos', 'pasivos'].forEach(k => { D[k] = num(data[k]) || ''; });
    return D;
  }

  function fieldHTML(f) {

    } else if (type === 'money') {
      input = '<input type="text" inputmode="numeric" data-k="' + k + '" data-money="1" value="' + esc(money(num(v))) + '" placeholder="0">';
    } else {
      input = '<input type="' + type + '" data-k="' + k + '" value="' + esc(v) + '"' + (type === 'number' ? ' min="0"' : '') + '>';
    }
    return '<div class="' + cls + '" data-wrap="' + k + '"><label>' + lab + req + '</label>' + input + (opt.hint ? '<div class="hint">' + opt.hint + '</div>' : '') + '</div>';
  }

    h += '<p class="section-hint" style="margin-top:18px">¿Quieres comparar otra entidad? Vuelve al paso «CDT», elige otra y regresa aquí: tus datos se conservan.</p>';
    return h;

  function render() {
    renderSteps();

    if (s.id === 'cdt') body.innerHTML = cdtHTML();
    else if (s.id === 'resumen') body.innerHTML = resumenHTML();
    else if (s.id === 'docs') body.innerHTML = docsHTML();
    else body.innerHTML = formHTML(s);
    $('progressFill').style.width = Math.round(((current + 1) / STEPS.length) * 100) + '%';
    $('btnBack').style.visibility = current === 0 ? 'hidden' : 'visible';
    $('btnNext').style.display = current === STEPS.length - 1 ? 'none' : '';
    $('btnNext').textContent = current === STEPS.length - 2 ? 'Generar formatos' : 'Siguiente';
    $('errMsg').textContent = '';
    refreshCalc();
  }

  /* ---------- Validación ---------- */
  function missing(stepIdx) {
    const s = STEPS[stepIdx], miss = [];
    const need = (k, l) => { if (!data[k] || (k === 'valor' && !num(data[k])) || (k === 'plazo_dias' && !num(data[k]))) miss.push([k, l]); };
    if (s.id === 'cdt') {
      if (!data.entidad) miss.push(['entidad', 'la entidad']);
      need('tipo_cdt', 'el tipo de CDT'); need('ciudad', 'la ciudad'); need('fecha_apertura', 'la fecha de apertura');
      need('plazo_dias', 'el plazo'); need('valor', 'el valor de la inversión');
    }
    if (s.sections) s.sections.forEach(sec => sec.f.forEach(f => { if (f[3] && f[3].req) need(f[0], f[1].toLowerCase()); }));
    return miss;
  }
  function tryGo(i) {
    if (i > current) {
      for (let j = current; j < i && j < STEPS.length; j++) {
        const m = missing(j);
        if (m.length) {
          if (j !== current) { current = j; render(); }
          m.forEach(x => { const w = document.querySelector('[data-wrap="' + x[0] + '"]'); if (w) w.classList.add('err'); });
          $('errMsg').textContent = 'Falta completar: ' + m.map(x => x[1]).join(', ') + '.';
          return;
        }
      }
    }
    current = Math.max(0, Math.min(i, STEPS.length - 1));
    render();
    window.scrollTo({ top: 0 });
  }

    if (!t) return;
    if (t.dataset.go != null) return tryGo(+t.dataset.go);
    if (t.dataset.ent) { data.entidad = t.dataset.ent; document.querySelectorAll('.ent-card').forEach(c => c.classList.toggle('sel', c.dataset.ent === data.entidad)); refreshCalc(); return; }
    if (t.dataset.plazo) { data.plazo_dias = t.dataset.plazo; const i = document.querySelector('[data-k="plazo_dias"]'); if (i) i.value = data.plazo_dias; refreshCalc(); return; }
    if (t.id === 'btnDemo') { fillDemo(); render(); return; }
    if (t.id === 'btnAll') { downloadAll(); return; }
    if (t.dataset.act) { const card = t.closest('.doc-card'); runDoc(card, t.dataset.act); }
  });

  /* ---------- Generación de PDF ---------- */
  const b64 = s => Uint8Array.from(atob(s), c => c.charCodeAt(0));
  const clean = s => String(s == null ? '' : s).replace(/[^\u0000-ÿ]/g, '?');
  const same = (a, b) => String(a == null ? '' : a).trim().toLowerCase().replace(/o$/, 'a') === String(b).toLowerCase().replace(/o$/, 'a');
  const dateOut = (iso2, part) => {
    if (!iso2) return '';
    const [y, m, d] = iso2.split('-');
    return part === 'd' ? d : part === 'm' ? m : part === 'y' ? y : d + '/' + m + '/' + y;
  };

  // Algunos PDF traen dos widgets casi idénticos para el mismo campo de texto; al rellenar se ve el texto doble.
  function dedupeWidgets(pdf, form) {
    const L = PDFLib;
    form.getFields().forEach(f => {
      if (!(f instanceof L.PDFTextField)) return;
      const af = f.acroField, ws = af.getWidgets();
      if (ws.length < 2) return;
      const kids = af.Kids();
      for (let i = ws.length - 1; i >= 1; i--) {
        const r = ws[i].getRectangle();
        for (let j = 0; j < i; j++) {
          const q = ws[j].getRectangle();
          if (Math.abs(r.x - q.x) < 3 && Math.abs(r.y - q.y) < 3 && Math.abs(r.width - q.width) < 3 && Math.abs(r.height - q.height) < 3) {
            const ref = String(kids.get(i));
            pdf.getPages().forEach(p => {
              const an = p.node.Annots(); if (!an) return;
              for (let k = an.size() - 1; k >= 0; k--) if (String(an.get(k)) === ref) an.remove(k);
            });
            af.removeWidget(i);
            break;
          }
        }
      }
    });
  }

  async function generate(docKey) {
    const L = PDFLib;
    const pdf = await L.PDFDocument.load(b64(PDFS[docKey]), { ignoreEncryption: true });
    const form = pdf.getForm();
    dedupeWidgets(pdf, form);
    const byName = {};
    form.getFields().forEach(f => { byName[f.getName()] = f; });
    const report = { notFound: [], skipped: [], text: 0, checks: 0 };

    // 1) Limpiar valores que el PDF trae marcados de fábrica
    Object.values(byName).forEach(f => {
      try {
        if (f instanceof L.PDFTextField) f.setText(undefined);
        else if (f instanceof L.PDFCheckBox) f.uncheck();
        else if (f instanceof L.PDFRadioGroup) f.clear();
        else if (f instanceof L.PDFDropdown || f instanceof L.PDFOptionList) f.clear();
      } catch (e) { /* campo sin valor */ }
    });

    // 2) Aplicar el mapeo
    const D = derive();
    const setText = (name, val) => {
      const f = byName[name];
      if (!f) { report.notFound.push(name); return; }
      if (val === '' || val == null) return;
      try {
        if (f instanceof L.PDFTextField) { f.setText(clean(val)); report.text++; }
        else report.skipped.push(name + ' (no es texto)');
      } catch (e) { report.skipped.push(name + ': ' + e.message); }
    };
    const setCheck = name => {
      const f = byName[name];
      if (!f) { report.notFound.push(name); return; }
      try {
        if (f instanceof L.PDFCheckBox) { f.check(); report.checks++; }
        else if (f instanceof L.PDFRadioGroup) { const o = f.getOptions(); if (o.length) { f.select(o[0]); report.checks++; } }
        else if (f instanceof L.PDFTextField) { f.setText('X'); report.checks++; }
        else report.skipped.push(name + ' (no es casilla)');
      } catch (e) { report.skipped.push(name + ': ' + e.message); }
    };
    (MAPPING[docKey] || []).forEach(r => {
      if (r.k === 't') {
        const v = D[r.key];
        setText(r.f, r.fmt === 'money' ? money(v) : (v == null ? '' : String(v)));
      } else if (r.k === 'c') { if (D[r.key] && same(D[r.key], r.val)) setCheck(r.f); }
      else if (r.k === 'x') setCheck(r.f);
      else if (r.k === 's') setText(r.f, r.val);
      else if (r.k === 'd') setText(r.f, dateOut(D[r.key], r.part));
      else if (r.k === 'tv') { if (D[r.cond[0]] === r.cond[1]) setText(r.f, r.fmt === 'money' ? money(D[r.key]) : D[r.key]); }
    });

    const font = await pdf.embedFont(L.StandardFonts.Helvetica);
    form.updateFieldAppearances(font);
    const bytes = await pdf.save();
    return { bytes, report };
  }

  function saveBlob(bytes, name, open) {
    const url = URL.createObjectURL(new Blob([bytes], { type: 'application/pdf' }));
    if (open) { window.open(url, '_blank'); return; }
    const a = document.createElement('a'); a.href = url; a.download = name;
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 4000);
  }
  async function runDoc(card, act) {
    const k = card.dataset.doc, st = card.querySelector('[data-st]');
    st.className = 'st'; st.textContent = 'Generando…';

    } catch (e) { st.className = 'st bad'; st.textContent = 'Error al generar: ' + e.message; console.error(e); }
  }
